Aim: Create a first ASP.NET Website and execute it on a web browser.

<strong><%= DateTime.Now.ToString("dd MMM yyyy HH:mm:ss") %></strong>

Steps to Create:
1. Open Visual Studio → New Project → ASP.NET Web Application (.NET Framework).
2. Select 'Web Forms' template and click OK.
3. Replace Default.aspx content with the code above.
4. Press F5 or Ctrl+F5 to run the application in the browser.
Output: Browser opens showing 'Welcome to My First ASP.NET Website!'
with current date and time displayed.

Practical 9: File Upload to Web Server
Aim: Create ASP.NET Web page that will upload any file to web server.
Default.aspx
<form id="form1" runat="server" enctype="multipart/form-data">
<div style="margin:40px;font-family:Arial;">
<h2>File Upload</h2>
<asp:FileUpload ID="fuFile" runat="server" Width="350px"/>
&nbsp;
<asp:Button ID="btnUpload" runat="server" Text="Upload File"
OnClick="btnUpload_Click"/>
<br/><br/>
<asp:Label ID="lblStatus" runat="server"/>
</div>
</form>
Default.aspx.cs
protected void btnUpload_Click(object sender, EventArgs e) {
if (fuFile.HasFile) {
try {
string fileName = System.IO.Path.GetFileName(fuFile.FileName);
string savePath = Server.MapPath("~/Uploads/") + fileName;
// Create Uploads directory if not exists
if (!System.IO.Directory.Exists(Server.MapPath("~/Uploads/")))
System.IO.Directory.CreateDirectory(Server.MapPath("~/Uploads/"));
fuFile.SaveAs(savePath);
lblStatus.Text = $"File '{fileName}' uploaded successfully!";
lblStatus.ForeColor = System.Drawing.Color.Green;
} catch (Exception ex) {
lblStatus.Text = "Upload failed: " + ex.Message;
lblStatus.ForeColor = System.Drawing.Color.Red;
}
} else {
lblStatus.Text = "Please select a file to upload.";
lblStatus.ForeColor = System.Drawing.Color.Orange;
}
}
Note: Create an 'Uploads' folder in your project root and set write permissions.
Output: File 'sample.pdf' uploaded successfully!

protected void btnPlus_Click(object sender, EventArgs e) {

int price = ViewState["Price"] != null
? (int)ViewState["Price"] : 0;
lblTotal.Text = $"Total Payment: Rs. {qty * price}";
}

void LoadCachedTime() {
if (Cache["CachedDateTime"] != null) {
lblCachedTime.Text = "Cached Time: "
+ Cache["CachedDateTime"].ToString()
+ " (from cache)";
} else {
string currentTime = DateTime.Now.ToString("dd-MMM-yyyy HH:mm:ss");
// Absolute expiration - cache for 60 seconds
Cache.Insert("CachedDateTime", currentTime, null,
DateTime.Now.AddSeconds(60),
System.Web.Caching.Cache.NoSlidingExpiration);
lblCachedTime.Text = "Fresh Time: " + currentTime + " (just cached)";
}
}
First load: Fresh Time (just cached). Next 60s: same time from cache. After 60s: new
time.

using System.Xml;
protected void btnCreate_Click(object sender, EventArgs e) {
XmlDocument doc = new XmlDocument();
XmlDeclaration dec = doc.CreateXmlDeclaration("1.0","utf-8",null);
doc.AppendChild(dec);
XmlElement root = doc.CreateElement("Students");
doc.AppendChild(root);
string[,] data = {
{"101","Rahul Sharma","BCA","5"},
{"102","Priya Patel","BCA","5"},
{"103","Amit Shah","BCA","3"}
};
for (int i = 0; i < data.GetLength(0); i++) {
XmlElement stu = doc.CreateElement("Student");
void Add(string tag, string val) {
XmlElement el = doc.CreateElement(tag);
el.InnerText = val; stu.AppendChild(el);
}
Add("StudentID", data[i,0]);
Add("Name", data[i,1]);
Add("Branch", data[i,2]);
Add("Semester", data[i,3]);
root.AppendChild(stu);
}
doc.Save(Server.MapPath("~/App_Data/students.xml"));
lblMsg.Text = "XML file created successfully!";
}

using System.Data; using System.Data.SqlClient;
using System.Xml; using System.Configuration;
protected void btnExport_Click(object sender, EventArgs e) {
// 1. Read from Database
string cs = ConfigurationManager.ConnectionStrings["MyDB"].ConnectionString;
DataTable dt = new DataTable();
using (SqlConnection conn = new SqlConnection(cs)) {
SqlDataAdapter da = new SqlDataAdapter(
"SELECT StudentID,Name,Branch,Semester,Email FROM Students", conn);
da.Fill(dt);
}
// 2. Write to XML
string xmlPath = Server.MapPath("~/App_Data/ExportedStudents.xml");
XmlDocument doc = new XmlDocument();
doc.AppendChild(doc.CreateXmlDeclaration("1.0","utf-8",null));
XmlElement root = doc.CreateElement("Students");
doc.AppendChild(root);
foreach (DataRow row in dt.Rows) {
XmlElement stu = doc.CreateElement("Student");
foreach (DataColumn col in dt.Columns) {
XmlElement el = doc.CreateElement(col.ColumnName);
el.InnerText = row[col].ToString();
stu.AppendChild(el);
}
root.AppendChild(stu);
}
doc.Save(xmlPath);
lblMsg.ForeColor = System.Drawing.Color.Green;
lblMsg.Text = $"{dt.Rows.Count} student records exported to XML successfully!";
}
Output: 25 student records exported to XML successfully!
